cmake_minimum_required(VERSION 3.21)
project(blacksmithzero LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenMP REQUIRED)
set(CORE "${CMAKE_CURRENT_SOURCE_DIR}/../../src/blacksmith-core")
file(GLOB SKILLS CONFIGURE_DEPENDS "${CORE}/src/skill-system/*.cpp")
add_library(blacksmithzero SHARED arena.cpp "${CORE}/src/domain/models.cpp" ${SKILLS})
target_include_directories(blacksmithzero PRIVATE "${CORE}/include")
target_link_libraries(blacksmithzero PRIVATE OpenMP::OpenMP_CXX)
set_target_properties(blacksmithzero PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
add_executable(predict predict.cpp)
add_executable(test-agent test-agent.cpp "${CORE}/src/domain/models.cpp" ${SKILLS})
target_include_directories(test-agent PRIVATE "${CORE}/include")
enable_testing()
add_test(NAME agent-schema COMMAND test-agent)
foreach(target blacksmithzero predict test-agent)
  set_target_properties(${target} PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
  foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(${target} PROPERTIES
      RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}"
      LIBRARY_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}")
  endforeach()
  if(WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_link_options(${target} PRIVATE -static)
    if(target STREQUAL "blacksmithzero")
      set_target_properties(${target} PROPERTIES PREFIX "")
    endif()
  endif()
endforeach()
